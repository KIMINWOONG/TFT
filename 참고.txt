-- 1. 학술대회 메인 정보 테이블
CREATE TABLE IF NOT EXISTS g5_conference (
    sy_id int(11) NOT NULL AUTO_INCREMENT COMMENT '집담회 ID',
    sy_title varchar(255) NOT NULL DEFAULT '' COMMENT '집담회 제목',
    sy_year year NOT NULL COMMENT '개최년도',
    sy_main_image varchar(255) DEFAULT '' COMMENT '메인 이미지 경로',
    sy_greeting longtext COMMENT '인사말 내용',
    sy_map_image varchar(255) DEFAULT '' COMMENT '약도 이미지 경로',
    sy_address varchar(500) NOT NULL DEFAULT '' COMMENT '주소',
    sy_time varchar(500) NOT NULL DEFAULT '' COMMENT '일시',
    sy_status varchar(20) NOT NULL DEFAULT 'planning' COMMENT '상태 (planning:기획중, active:진행중, completed:완료)',
    sy_reg_date datetime NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '등록일',
    sy_update_date datetime NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '수정일',
    PRIMARY KEY (sy_id),
    KEY idx_year (sy_year),
    KEY idx_status (sy_status)
) ENGINE=InnoDB DEFAULT CHARSET=utf8 COMMENT='학술대회 집담회 메인';

-- 2. 학술대회 일정 테이블
CREATE TABLE IF NOT EXISTS g5_conference_schedule (
    ss_id int(11) NOT NULL AUTO_INCREMENT COMMENT '일정 ID',
    ss_sy_id int(11) NOT NULL COMMENT '집담회 ID',
    ss_time varchar(255) NOT NULL COMMENT '시간',
    ss_title varchar(500) NOT NULL DEFAULT '' COMMENT '일정 제목',
    ss_speaker varchar(100) DEFAULT '' COMMENT '발표자명',
    ss_order int(11) NOT NULL DEFAULT 0 COMMENT '순서',
    ss_type varchar(10) DEFAULT '' COMMENT '일정 유형',
    PRIMARY KEY (ss_id),
    KEY idx_symposium_id (ss_sy_id),
    KEY idx_order (ss_order),
    FOREIGN KEY (ss_sy_id) REFERENCES g5_conference(sy_id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8 COMMENT='학술대회 일정';

-- 3. 학술대회 연자 정보 테이블
CREATE TABLE IF NOT EXISTS g5_conference_speaker (
    sp_id int(11) NOT NULL AUTO_INCREMENT COMMENT '연자 ID',
    sp_sy_id int(11) NOT NULL COMMENT '집담회 ID',
    sp_photo varchar(255) DEFAULT '' COMMENT '사진 경로',
    sp_name varchar(100) NOT NULL DEFAULT '' COMMENT '연자명',
    sp_specialty varchar(500) DEFAULT '' COMMENT '전문분야',
    sp_order int(11) NOT NULL DEFAULT 0 COMMENT '순서',
    sp_reg_date datetime NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '등록일',
    PRIMARY KEY (sp_id),
    KEY idx_symposium_id (sp_sy_id),
    KEY idx_order (sp_order),
    FOREIGN KEY (sp_sy_id) REFERENCES g5_conference(sy_id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8 COMMENT='학술대회 연자 정보';
